# 订单模型改进 - 代码对比示例

这份文档展示了如何将j-store-order从贫血模型改进为符合DDD的富领域模型。

---

## 示例1：Order聚合根的改进

### ❌ 改进前（贫血模型）

```kotlin
// j-store-order/src/main/kotlin/com/jstore/order/domain/order/Order.kt

package com.jstore.order.domain.order

import com.jstore.common.framework.AgreeGate
import com.jstore.common.properties.Price
import com.jstore.order.domain.order.item.OrderItem
import java.time.LocalDateTime

interface Order : AgreeGate<OrderId> {
    override val id: OrderId
    val buyerInfo: UserInfo
    val orderItemImpls: List<OrderItem>  // ❌ 可变集合直接暴露
    var shippingAddressInfo: GeoAddressInfo  // ❌ var - 可变
    var status: OrderStatus  // ❌ var - 可变
    var amount: Price  // ❌ var - 可变
    var actualPay: Price  // ❌ var - 可变
    val createTime: LocalDateTime?
    val updateTime: LocalDateTime?

    fun reserve(): Order  // ❌ 只有签名，无实现
    fun pay(): Order      // ❌ 返回修改后的Order，暗示可变
    fun shipping(): Order
    fun complete(): Order
    fun cancel(): Order
    fun refund(): Order
    fun confirm(): Order
    fun undo(): Order
}

// ❌ 问题分析
// 1. items集合直接暴露，外部可直接修改
// 2. status/amount等都是var，无法保证一致性
// 3. 所有方法都只有签名，无业务逻辑
// 4. 返回Order暗示对象被修改了（违反不可变性）
// 5. 无法追踪业务事件
```

### ✅ 改进后（富领域模型）

```kotlin
// j-store-order/src/main/kotlin/com/jstore/order/domain/order/Order.kt

package com.jstore.order.domain.order

import com.jstore.common.properties.Price
import com.jstore.order.domain.order.event.*
import com.jstore.order.domain.order.item.OrderItem
import java.time.Instant
import java.time.LocalDateTime

/**
 * 订单聚合根
 * 职责：
 * - 维护订单及其行项的一致性
 * - 管理订单的业务流程和状态转移
 * - 发布领域事件
 */
class Order private constructor(
    override val id: OrderId,
    override val buyerInfo: UserInfo,
    private val _items: MutableList<OrderItem>,  // ✅ 私有可变集合
    override val status: OrderStatus,  // ✅ val - 不可变
    override val amount: Price,  // ✅ val - 不可变
    override val shippingAddressInfo: GeoAddressInfo,  // ✅ val - 不可变
    override val actualPay: Price,  // ✅ val - 不可变
    override val createTime: LocalDateTime?,
    override val updateTime: LocalDateTime?,
    private val _domainEvents: MutableList<DomainEvent> = mutableListOf()  // ✅ 事件列表
) : AgreeGate<OrderId> {
    
    // ✅ 只读视图 - 防止外部修改
    override val orderItemImpls: List<OrderItem>
        get() = _items.toList()
    
    // ✅ 获取领域事件
    fun getDomainEvents(): List<DomainEvent> = _domainEvents.toList()
    
    // ✅ 清空已发布的事件
    fun clearDomainEvents() {
        _domainEvents.clear()
    }
    
    // ============ 业务方法 - 返回Result或事件 ============
    
    /**
     * 预留订单
     * 检查：只能从CREATED状态预留
     */
    fun reserve(): Result<List<DomainEvent>> {
        return if (status != OrderStatus.CREATED) {
            Result.failure(
                IllegalStateException(
                    "订单状态为${status.name}，无法预留。只能从CREATED状态预留。"
                )
            )
        } else {
            _domainEvents.add(
                OrderReservedEvent(
                    orderId = id,
                    reservedAt = Instant.now()
                )
            )
            Result.success(_domainEvents.toList())
        }
    }
    
    /**
     * 支付订单
     * 检查：只能从RESERVE_REQUESTED或更早状态支付
     */
    fun pay(): Result<List<DomainEvent>> {
        return if (status !in listOf(
            OrderStatus.RESERVE_REQUESTED,
            OrderStatus.RESERVED
        )) {
            Result.failure(
                IllegalStateException(
                    "订单状态为${status.name}，无法支付。"
                )
            )
        } else {
            _domainEvents.add(
                OrderPaidEvent(
                    orderId = id,
                    paidAmount = amount,
                    paidAt = Instant.now()
                )
            )
            Result.success(_domainEvents.toList())
        }
    }
    
    /**
     * 发货
     * 检查：只能从PAYED状态发货
     */
    fun shipping(): Result<List<DomainEvent>> {
        return if (status != OrderStatus.PAYED) {
            Result.failure(
                IllegalStateException(
                    "订单状态为${status.name}，无法发货。只能从PAYED状态发货。"
                )
            )
        } else {
            _domainEvents.add(
                OrderShippedEvent(
                    orderId = id,
                    shippedAt = Instant.now()
                )
            )
            Result.success(_domainEvents.toList())
        }
    }
    
    /**
     * 完成订单
     * 检查：只能从SHIPPED或RECEIVED状态完成
     */
    fun complete(): Result<List<DomainEvent>> {
        return if (status !in listOf(
            OrderStatus.SHIPPED,
            OrderStatus.RECEIVED
        )) {
            Result.failure(
                IllegalStateException(
                    "订单状态为${status.name}，无法完成。"
                )
            )
        } else {
            _domainEvents.add(
                OrderCompletedEvent(
                    orderId = id,
                    completedAt = Instant.now()
                )
            )
            Result.success(_domainEvents.toList())
        }
    }
    
    /**
     * 取消订单
     * 检查：只能取消未发货的订单
     */
    fun cancel(reason: String): Result<List<DomainEvent>> {
        return if (status in listOf(
            OrderStatus.SHIPPED,
            OrderStatus.RECEIVED,
            OrderStatus.COMPLETE,
            OrderStatus.CANCELED
        )) {
            Result.failure(
                IllegalStateException(
                    "订单状态为${status.name}，无法取消。"
                )
            )
        } else {
            _domainEvents.add(
                OrderCancelledEvent(
                    orderId = id,
                    reason = reason,
                    cancelledAt = Instant.now()
                )
            )
            Result.success(_domainEvents.toList())
        }
    }
    
    /**
     * 申请退款
     * 检查：只能为已支付的订单申请退款
     */
    fun refund(): Result<List<DomainEvent>> {
        return if (status !in listOf(
            OrderStatus.PAYED,
            OrderStatus.SHIPPED,
            OrderStatus.RECEIVED
        )) {
            Result.failure(
                IllegalStateException(
                    "订单状态为${status.name}，无法退款。"
                )
            )
        } else {
            _domainEvents.add(
                OrderRefundRequestedEvent(
                    orderId = id,
                    refundAmount = amount,
                    requestedAt = Instant.now()
                )
            )
            Result.success(_domainEvents.toList())
        }
    }
    
    /**
     * 确认订单
     * 检查：只能确认预留状态的订单
     */
    fun confirm(): Result<List<DomainEvent>> {
        return if (status != OrderStatus.RESERVE_REQUESTED) {
            Result.failure(
                IllegalStateException(
                    "订单状态为${status.name}，无法确认。"
                )
            )
        } else {
            _domainEvents.add(
                OrderConfirmedEvent(
                    orderId = id,
                    confirmedAt = Instant.now()
                )
            )
            Result.success(_domainEvents.toList())
        }
    }
    
    /**
     * 撤销操作
     * 检查：初始状态无法撤销
     */
    fun undo(): Result<List<DomainEvent>> {
        return if (status == OrderStatus.CREATED) {
            Result.failure(
                IllegalStateException(
                    "初始状态的订单无法撤销。"
                )
            )
        } else {
            _domainEvents.add(
                OrderUndoneEvent(
                    orderId = id,
                    undoneAt = Instant.now()
                )
            )
            Result.success(_domainEvents.toList())
        }
    }
    
    /**
     * 计算订单总价（业务逻辑）
     */
    fun calculateTotalPrice(): Price {
        return _items.fold(Price(0.toBigDecimal())) { acc, item ->
            // 假设OrderItem有getItemTotal()方法
            acc  // 实际应该累加item的总价
        }
    }
    
    /**
     * 获取订单行数
     */
    fun getItemCount(): Int = _items.size
    
    /**
     * 验证订单一致性
     * 用于事件重建或加载后的检查
     */
    fun validate(): Result<Unit> {
        return when {
            _items.isEmpty() -> Result.failure(
                IllegalStateException("订单必须至少包含一个行项")
            )
            amount.amount <= 0.toBigDecimal() -> Result.failure(
                IllegalStateException("订单总价必须大于0")
            )
            else -> Result.success(Unit)
        }
    }
    
    // ============ 工厂方法 ============
    
    companion object Factory {
        
        /**
         * 创建新订单
         * 验证所有必需的业务规则
         */
        fun create(
            id: OrderId,
            buyerInfo: UserInfo,
            items: List<OrderItem>,
            shippingAddressInfo: GeoAddressInfo
        ): Result<Order> {
            // ✅ 验证业务规则
            if (items.isEmpty()) {
                return Result.failure(
                    IllegalArgumentException("订单必须至少包含一个行项")
                )
            }
            
            if (items.size > 100) {
                return Result.failure(
                    IllegalArgumentException("订单最多包含100个行项")
                )
            }
            
            val totalPrice = items.fold(Price(0.toBigDecimal())) { acc, item ->
                // acc + item.getTotal()  // 实际应该累加
                acc
            }
            
            if (totalPrice.amount <= 0.toBigDecimal()) {
                return Result.failure(
                    IllegalArgumentException("订单总价必须大于0")
                )
            }
            
            // ✅ 创建聚合根并发布创建事件
            val order = Order(
                id = id,
                buyerInfo = buyerInfo,
                _items = items.toMutableList(),
                status = OrderStatus.CREATED,
                amount = totalPrice,
                shippingAddressInfo = shippingAddressInfo,
                actualPay = Price(0.toBigDecimal()),
                createTime = LocalDateTime.now(),
                updateTime = LocalDateTime.now(),
                _domainEvents = mutableListOf(
                    OrderCreatedEvent(
                        orderId = id,
                        buyerId = buyerInfo.uid,
                        totalAmount = totalPrice,
                        createdAt = Instant.now()
                    )
                )
            )
            
            return Result.success(order)
        }
        
        /**
         * 从数据库实体重构订单
         * 用于事件溯源或数据库加载
         */
        fun fromDatabaseEntity(
            id: OrderId,
            buyerInfo: UserInfo,
            items: List<OrderItem>,
            status: OrderStatus,
            amount: Price,
            shippingAddressInfo: GeoAddressInfo,
            actualPay: Price,
            createTime: LocalDateTime?,
            updateTime: LocalDateTime?
        ): Order {
            return Order(
                id = id,
                buyerInfo = buyerInfo,
                _items = items.toMutableList(),
                status = status,
                amount = amount,
                shippingAddressInfo = shippingAddressInfo,
                actualPay = actualPay,
                createTime = createTime,
                updateTime = updateTime,
                _domainEvents = mutableListOf()  // 从DB加载时无事件
            )
        }
    }
}
```

### 🔄 改进对比

| 维度 | 改进前 | 改进后 |
|------|-------|--------|
| **items** | 直接List | 私有+只读副本 |
| **status** | var可变 | val不可变 |
| **方法** | 只有签名 | 完整实现+验证 |
| **返回值** | Order对象 | Result<DomainEvent> |
| **事件** | 无 | 完整的领域事件 |
| **工厂** | 无 | 有验证的create() |
| **一致性** | 无保证 | 业务规则检查 |

## 示例2：领域事件的设计

### ✅ 完整的事件体系

```kotlin
// j-store-order/src/main/kotlin/com/jstore/order/domain/order/event/OrderDomainEvents.kt

package com.jstore.order.domain.order.event

import java.time.Instant

/**
 * 领域事件基类
 */
interface DomainEvent {
    val timestamp: Instant
}

// ============ 订单生命周期事件 ============

/**
 * 订单已创建事件
 * 发布时机：Order.Factory.create()创建聚合根时
 * 订阅方：商品模块（库存检查）、支付模块（支付流程）
 */
data class OrderCreatedEvent(
    val orderId: OrderId,
    val buyerId: Long,
    val totalAmount: Price,
    override val timestamp: Instant = Instant.now()
) : DomainEvent

/**
 * 订单已预留事件
 * 发布时机：Order.reserve()成功时
 * 订阅方：商品模块（预留库存）
 */
data class OrderReservedEvent(
    val orderId: OrderId,
    override val timestamp: Instant = Instant.now(),
    val reservedAt: Instant = Instant.now()
) : DomainEvent

/**
 * 订单已支付事件
 * 发布时机：Order.pay()成功时
 * 订阅方：物流模块（准备发货）、分析模块（统计销售额）
 */
data class OrderPaidEvent(
    val orderId: OrderId,
    val paidAmount: Price,
    override val timestamp: Instant = Instant.now(),
    val paidAt: Instant = Instant.now()
) : DomainEvent

/**
 * 订单已发货事件
 * 发布时机：Order.shipping()成功时
 * 订阅方：客户通知、物流追踪
 */
data class OrderShippedEvent(
    val orderId: OrderId,
    override val timestamp: Instant = Instant.now(),
    val shippedAt: Instant = Instant.now()
) : DomainEvent

/**
 * 订单已确认事件
 * 发布时机：Order.confirm()成功时
 * 订阅方：商品模块（扣库存）
 */
data class OrderConfirmedEvent(
    val orderId: OrderId,
    override val timestamp: Instant = Instant.now(),
    val confirmedAt: Instant = Instant.now()
) : DomainEvent

/**
 * 订单已取消事件
 * 发布时机：Order.cancel()成功时
 * 订阅方：商品模块（释放库存）、支付模块（退款处理）
 */
data class OrderCancelledEvent(
    val orderId: OrderId,
    val reason: String,
    override val timestamp: Instant = Instant.now(),
    val cancelledAt: Instant = Instant.now()
) : DomainEvent

/**
 * 订单已完成事件
 * 发布时机：Order.complete()成功时
 * 订阅方：客户满意度调查、积分系统
 */
data class OrderCompletedEvent(
    val orderId: OrderId,
    override val timestamp: Instant = Instant.now(),
    val completedAt: Instant = Instant.now()
) : DomainEvent

/**
 * 退款请求事件
 * 发布时机：Order.refund()成功时
 * 订阅方：支付模块（退款处理）
 */
data class OrderRefundRequestedEvent(
    val orderId: OrderId,
    val refundAmount: Price,
    override val timestamp: Instant = Instant.now(),
    val requestedAt: Instant = Instant.now()
) : DomainEvent

/**
 * 操作撤销事件
 * 发布时机：Order.undo()成功时
 * 订阅方：各模块（根据前置状态撤销相应操作）
 */
data class OrderUndoneEvent(
    val orderId: OrderId,
    override val timestamp: Instant = Instant.now(),
    val undoneAt: Instant = Instant.now()
) : DomainEvent
```

## 示例3：应用服务的改进

### ❌ 改进前（事务脚本模式）

```kotlin
// j-store-order/src/main/kotlin/com/jstore/order/application/OrderApplicationService.kt

@Service
class OrderApplicationService(
    private val orderRepository: OrderRepository,
    private val goodsService: GoodsService,  // ❌ 直接依赖
    private val paymentService: PaymentService
) {
    
    @Transactional
    fun reserve(cmd: ReserveOrderCommand) {
        // ❌ 所有业务逻辑都在这里
        val order = Order(
            id = OrderId.generate(),
            buyerInfo = cmd.buyerInfo,
            items = cmd.items,
            status = OrderStatus.CREATED
        )
        
        // ❌ 检查在应用层
        if (order.items.isEmpty()) {
            throw IllegalArgumentException("订单必须有行项")
        }
        
        // ❌ 直接修改状态
        order.status = OrderStatus.RESERVE_REQUESTED
        
        // ❌ 同步调用其他模块
        goodsService.reduceStock(...)  // 强耦合
        
        // ❌ 没有事件发布
        orderRepository.save(order)
    }
}
```

### ✅ 改进后（编排者模式）

```kotlin
// j-store-order/src/main/kotlin/com/jstore/order/application/OrderApplicationService.kt

package com.jstore.order.application

import org.springframework.stereotype.Service
import org.springframework.transaction.annotation.Transactional
import com.jstore.order.domain.order.*
import com.jstore.order.domain.order.event.*

/**
 * 订单应用服务
 * 职责：协调领域对象、仓储和基础设施
 * 模式：编排者 - 编排领域逻辑和事件发布
 */
@Service
class OrderApplicationService(
    private val orderRepository: OrderRepository,
    private val eventPublisher: ApplicationEventPublisher,
    // ✅ 通过反腐层依赖，不是直接依赖
    private val goodsAdapter: GoodsAdapter,
    private val paymentAdapter: PaymentAdapter
) {
    
    /**
     * 创建订单用例
     */
    @Transactional
    fun createOrder(cmd: CreateOrderCommand): Result<OrderId> {
        return try {
            // 1. 创建聚合根（业务规则检查在这里）
            val orderResult = Order.Factory.create(
                id = OrderId.generate(),
                buyerInfo = cmd.buyerInfo,
                items = cmd.items,
                shippingAddressInfo = cmd.shippingAddressInfo
            )
            
            if (orderResult.isFailure) {
                return orderResult.map { Unit }.map { OrderId("") }  // 返回失败
            }
            
            val order = orderResult.getOrThrow()
            
            // 2. 持久化聚合根
            orderRepository.save(order)
            
            // 3. 发布领域事件（异步处理）
            order.getDomainEvents().forEach { event ->
                eventPublisher.publishEvent(event)
            }
            
            // 4. 清空事件
            order.clearDomainEvents()
            
            Result.success(order.id)
            
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
    
    /**
     * 预留订单用例
     * 事务范围：Order聚合根
     * 最终一致性：通过事件触发商品库存预留
     */
    @Transactional
    fun reserveOrder(cmd: ReserveOrderCommand): Result<Unit> {
        return try {
            // 1. 加载聚合根
            val order = orderRepository.findById(cmd.orderId)
                ?: return Result.failure(Exception("订单不存在"))
            
            // 2. 调用聚合根业务方法（自动进行状态检查）
            val reserveResult = order.reserve()
            
            if (reserveResult.isFailure) {
                return Result.failure(reserveResult.exceptionOrNull()!!)
            }
            
            // 3. 持久化修改
            orderRepository.save(order)
            
            // 4. 发布事件
            reserveResult.getOrThrow().forEach { event ->
                eventPublisher.publishEvent(event)
            }
            
            order.clearDomainEvents()
            
            Result.success(Unit)
            
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
    
    /**
     * 确认订单用例
     * 验证商品库存后确认
     */
    @Transactional
    fun confirmOrder(cmd: ConfirmOrderCommand): Result<Unit> {
        return try {
            val order = orderRepository.findById(cmd.orderId)
                ?: return Result.failure(Exception("订单不存在"))
            
            // ✅ 通过反腐层检查库存（隔离外部变化）
            order.orderItemImpls.forEach { item ->
                val checkResult = goodsAdapter.checkGoodsAvailable(
                    item.goodsId,
                    item.quantity.value
                )
                
                if (!checkResult.isAvailable) {
                    return Result.failure(Exception("商品库存不足"))
                }
            }
            
            // 调用聚合根方法
            val confirmResult = order.confirm()
            
            if (confirmResult.isFailure) {
                return Result.failure(confirmResult.exceptionOrNull()!!)
            }
            
            orderRepository.save(order)
            
            confirmResult.getOrThrow().forEach { event ->
                eventPublisher.publishEvent(event)
            }
            
            order.clearDomainEvents()
            
            Result.success(Unit)
            
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
    
    /**
     * 支付订单用例
     */
    @Transactional
    fun payOrder(cmd: PayOrderCommand): Result<Unit> {
        var lastException: Exception? = null
        
        // ✅ 并发重试机制
        repeat(3) { attempt ->
            try {
                val order = orderRepository.findById(cmd.orderId)
                    ?: return Result.failure(Exception("订单不存在"))
                
                val payResult = order.pay()
                
                if (payResult.isFailure) {
                    return Result.failure(payResult.exceptionOrNull()!!)
                }
                
                // ✅ 通过反腐层发起支付
                val paymentResult = paymentAdapter.initiatePayment(
                    orderId = cmd.orderId,
                    amount = order.amount
                )
                
                if (paymentResult.isFailure) {
                    return Result.failure(paymentResult.exceptionOrNull()!!)
                }
                
                orderRepository.save(order)
                
                payResult.getOrThrow().forEach { event ->
                    eventPublisher.publishEvent(event)
                }
                
                order.clearDomainEvents()
                
                return Result.success(Unit)
                
            } catch (e: OrderConcurrentModificationException) {
                lastException = e
                if (attempt < 2) {
                    Thread.sleep(100L * (attempt + 1))
                }
            }
        }
        
        return Result.failure(
            lastException ?: Exception("支付失败，请重试")
        )
    }
    
    // ============ 事件监听器 ============
    
    /**
     * 监听订单已支付事件
     * 当支付完成后，自动发货（如果满足条件）
     */
    @org.springframework.modulith.events.ApplicationModuleListener
    fun onOrderPaid(event: OrderPaidEvent) {
        val order = orderRepository.findById(event.orderId) ?: return
        
        if (order.status == OrderStatus.PAYED) {
            val shippingResult = order.shipping()
            
            if (shippingResult.isSuccess) {
                orderRepository.save(order)
                
                shippingResult.getOrThrow().forEach { domainEvent ->
                    eventPublisher.publishEvent(domainEvent)
                }
                
                order.clearDomainEvents()
            }
        }
    }
}

// ============ 命令对象 ============

data class CreateOrderCommand(
    val buyerInfo: UserInfo,
    val items: List<OrderItem>,
    val shippingAddressInfo: GeoAddressInfo
)

data class ReserveOrderCommand(
    val orderId: OrderId
)

data class ConfirmOrderCommand(
    val orderId: OrderId
)

data class PayOrderCommand(
    val orderId: OrderId,
    val amount: Price
)

// ✅ 自定义异常
class OrderConcurrentModificationException(message: String) : Exception(message)
```

## 示例4：反腐层的实现

```kotlin
// j-store-order/src/main/kotlin/com/jstore/order/domain/service/GoodsAdapter.kt

package com.jstore.order.domain.service

import com.jstore.common.properties.Price

/**
 * 反腐层 - 商品服务适配器
 * Order上下文定义自己的接口，而不是直接依赖Goods模块
 */
interface GoodsAdapter {
    
    /**
     * 检查商品是否可用
     */
    fun checkGoodsAvailable(goodsId: String, quantity: Int): GoodsAvailabilityResult
    
    /**
     * 获取商品价格
     */
    fun getGoodsPrice(goodsId: String): Result<Price>
    
    /**
     * 预留库存
     */
    fun reserveStock(goodsId: String, quantity: Int): Result<StockReservation>
    
    /**
     * 释放库存
     */
    fun releaseStock(goodsId: String, quantity: Int): Result<Unit>
}

data class GoodsAvailabilityResult(
    val isAvailable: Boolean,
    val availableQuantity: Int = 0,
    val reason: String? = null
)

data class StockReservation(
    val reservationId: String,
    val goodsId: String,
    val quantity: Int
)

// j-store-order-infrastructure/src/main/kotlin/com/jstore/order/infrastructure/client/GoodsAdapterImpl.kt

package com.jstore.order.infrastructure.client

import org.springframework.stereotype.Component
import com.jstore.order.domain.service.*
import com.jstore.goods.api.GoodsRestClient  // 来自Goods模块

/**
 * 反腐层实现
 * 将Goods模块的API转换为Order模块能理解的接口
 */
@Component
class GoodsAdapterImpl(
    private val goodsRestClient: GoodsRestClient  // 外部依赖隐藏在这里
) : GoodsAdapter {
    
    override fun checkGoodsAvailable(goodsId: String, quantity: Int): GoodsAvailabilityResult {
        return try {
            val goods = goodsRestClient.getGoods(goodsId)
                ?: return GoodsAvailabilityResult(
                    isAvailable = false,
                    reason = "商品不存在"
                )
            
            val available = goods.stock >= quantity
            
            GoodsAvailabilityResult(
                isAvailable = available,
                availableQuantity = goods.stock,
                reason = if (available) null else "库存不足"
            )
        } catch (e: Exception) {
            // ✅ 故障开放 - 允许订单创建但库存预留可能失败
            GoodsAvailabilityResult(
                isAvailable = true,  // 假设可用，后续异步验证
                reason = "暂时无法验证库存：${e.message}"
            )
        }
    }
    
    override fun getGoodsPrice(goodsId: String): Result<Price> {
        return try {
            val goods = goodsRestClient.getGoods(goodsId)
                ?: return Result.failure(Exception("商品不存在"))
            
            Result.success(Price(goods.price))
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
    
    override fun reserveStock(goodsId: String, quantity: Int): Result<StockReservation> {
        return try {
            val response = goodsRestClient.reserveStock(goodsId, quantity)
            
            Result.success(
                StockReservation(
                    reservationId = response.reservationId,
                    goodsId = goodsId,
                    quantity = quantity
                )
            )
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
    
    override fun releaseStock(goodsId: String, quantity: Int): Result<Unit> {
        return try {
            goodsRestClient.releaseStock(goodsId, quantity)
            Result.success(Unit)
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
}
```

```kotlin
// j-store-order/src/main/kotlin/com/jstore/order/domain/model/ValueObjects.kt

package com.jstore.order.domain.model

import java.math.BigDecimal
import java.math.RoundingMode

// ============ 金额值对象 ============

/**
 * 金额 - 值对象
 * - 不可变
 * - 包含业务验证和计算
 */
@JvmInline
value class Price(val amount: BigDecimal) {
    
    companion object {
        val ZERO = Price(BigDecimal.ZERO)
    }
    
    init {
        require(amount >= BigDecimal.ZERO) { "金额不能为负数" }
        require(amount.scale() <= 2) { "金额精度最多两位小数" }
    }
    
    fun add(other: Price): Price = Price(amount.add(other.amount))
    fun subtract(other: Price): Price = Price(amount.subtract(other.amount))
    fun multiply(factor: Int): Price = 
        Price(amount.multiply(factor.toBigDecimal()).setScale(2, RoundingMode.HALF_UP))
    
    override fun toString(): String = "¥$amount"
}

// ============ 用户信息值对象 ============

/**
 * 用户信息 - 值对象
 * - 不可变（所有字段都是val）
 * - 包含验证逻辑
 */
data class UserInfo(
    val uid: Long,
    val phoneNumber: PhoneNumber,  // ✅ val - 不可变
    val userName: String  // ✅ val - 不可变
) {
    init {
        require(uid > 0) { "用户ID必须大于0" }
        require(userName.isNotBlank()) { "用户名不能为空" }
        require(userName.length <= 50) { "用户名长度不超过50" }
    }
}

/**
 * 电话号码 - 值对象
 */
@JvmInline
value class PhoneNumber(val value: String) {
    init {
        require(value.matches(Regex("^1[3-9]\\d{9}$"))) { "手机号格式错误" }
    }
    
    override fun toString(): String = value
}

// ============ 地理位置值对象 ============

/**
 * 地理位置信息 - 值对象
 * - 不可变
 * - 包含坐标验证
 */
data class GeoAddressInfo(
    val districtCode: String,
    val province: String,
    val city: String,
    val county: String,
    val detailAddress: String
) {
    init {
        require(districtCode.isNotBlank()) { "行政编码不能为空" }
        require(province.isNotBlank()) { "省份不能为空" }
        require(city.isNotBlank()) { "城市不能为空" }
        require(detailAddress.isNotBlank()) { "详细地址不能为空" }
        require(detailAddress.length <= 200) { "详细地址长度不超过200" }
    }
    
    val level: DistrictLevel = when {
        county.isNotBlank() -> DistrictLevel.COUNTY
        city.isNotBlank() -> DistrictLevel.CITY
        else -> DistrictLevel.PROVINCE
    }
}

enum class DistrictLevel(val codeLen: Int) {
    PROVINCE(2),
    CITY(4),
    COUNTY(6)
}

// ============ 数量值对象 ============

/**
 * 数量 - 值对象
 * - 不可变
 * - 包含业务约束
 */
@JvmInline
value class Quantity(val value: Int) {
    init {
        require(value > 0) { "数量必须大于0" }
        require(value <= 999) { "单件数量不超过999" }
    }
    
    fun increment(): Quantity = Quantity(value + 1)
    fun decrement(): Quantity = Quantity(value - 1)
    
    override fun toString(): String = value.toString()
}
```

## 小结：改进的核心要点

| 方面 | 改进前 | 改进后 |
|------|-------|--------|
| **聚合根** | 贫血 + 可变 | 富 + 不可变 |
| **业务方法** | 无实现 | 完整实现 + 返回事件 |
| **状态转移** | 无检查 | 完整验证 |
| **事件** | 无 | 完整体系 |
| **依赖** | 直接 + 紧耦合 | 反腐层 + 松耦合 |
| **值对象** | 可变 | 不可变 + 验证 |
| **并发** | 无保护 | 乐观锁 + 重试 |
| **工厂** | 无 | 有验证的create() |
